cmake_minimum_required(VERSION 3.20)
project(html_rdf_c C)
set(CMAKE_C_STANDARD 11)
find_package(CURL REQUIRED)
find_path(LEXBOR_INCLUDE_DIR lexbor/html/html.h REQUIRED)
find_library(LEXBOR_LIBRARY NAMES lexbor REQUIRED)
add_executable(html-rdf-c extract.c)
target_include_directories(html-rdf-c PRIVATE ${LEXBOR_INCLUDE_DIR})
target_link_libraries(html-rdf-c PRIVATE ${LEXBOR_LIBRARY} CURL::libcurl)
