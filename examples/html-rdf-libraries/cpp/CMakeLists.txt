cmake_minimum_required(VERSION 3.20)
project(html_rdf_cpp CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(CURL REQUIRED)
find_path(LEXBOR_INCLUDE_DIR lexbor/html/html.h REQUIRED)
find_library(LEXBOR_LIBRARY NAMES lexbor REQUIRED)
add_executable(html-rdf-cpp extract.cpp)
target_include_directories(html-rdf-cpp PRIVATE ${LEXBOR_INCLUDE_DIR})
target_link_libraries(html-rdf-cpp PRIVATE ${LEXBOR_LIBRARY} CURL::libcurl)
